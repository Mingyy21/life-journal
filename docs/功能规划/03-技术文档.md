### 1.1 insight 表新增字段

当前 insights 表结构不变，新增字段：

### 1.2 insight_relations 新表

```sql
CREATE TABLE IF NOT EXISTS insight_relations (
  id TEXT PRIMARY KEY,
  "userId" UUID NOT NULL DEFAULT auth.uid(),
  "sourceId" TEXT NOT NULL,       -- 源感悟 ID
  "targetId" TEXT NOT NULL,       -- 目标感悟 ID
  "relationType" TEXT NOT NULL CHECK ("relationType" IN ('parent', 'child', 'sibling', 'opposite')),
  "createdAt" TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 索引
CREATE INDEX IF NOT EXISTS idx_ir_sourceId ON insight_relations("userId", "sourceId");
CREATE INDEX IF NOT EXISTS idx_ir_targetId ON insight_relations("userId", "targetId");

-- RLS
ALTER TABLE insight_relations ENABLE ROW LEVEL SECURITY;
CREATE POLICY "owner_access" ON insight_relations FOR ALL USING (auth.uid() = "userId");
```

### 2.1 生成展望文章

```typescript
// 调用 /api/trend 类似的 API 路由
async function generateVision(topicId: string, diaries: Diary[]): Promise<{
  article: string;
  baselines: { topicName: string; suggestedBaseline: string }[];
}> {
  const prompt = `你是一位人生教练。基于用户以下关于未来生活的日记片段，做两件事：

1. 写一篇连贯、温暖、有画面感的人生展望文章（800-1200字），
   用第一人称"我"，描绘用户理想中的未来生活。

2. 从文章中提取需要面对的 3-6 个人生课题，
   为每个课题建议一条底线——即"在这个课题上，最低不能跌破什么"。

日记片段：
${diaries.map(d => `【${d.createdAt.toLocaleDateString()}】${d.content.slice(0, 500)}`).join('\n\n')}

返回格式（JSON）：
{
  "article": "完整的文章内容...",
  "baselines": [
    { "topicName": "工作", "suggestedBaseline": "不内耗，只做分内之事" },
    ...
  ]
}`;

  // 调用 DeepSeek API
  const result = await callDeepSeek(prompt);
  return JSON.parse(result);
}
```

### 2.2 课题匹配逻辑

```typescript
async function matchTopics(baselines: { topicName: string }[]): Promise<Map<string, string>> {
  const existingTopics = await db.topics.toArray();
  const match = new Map<string, string>(); // baselines.topicName → existing topic.id

  for (const bl of baselines) {
    const found = existingTopics.find(t =>
      t.name.includes(bl.topicName) || bl.topicName.includes(t.name)
    );
    if (found) {
      match.set(bl.topicName, found.id);
    } else {
      // 未匹配 → 弹窗让用户选择创建新课题或忽略
      match.set(bl.topicName, '__new__');
    }
  }
  return match;
}
```

```typescript
// ── 展望 CRUD ──

export async function getVisionInsight(topicId: string): Promise<Insight | undefined> {
  return db.insights.filter(i => i.linkedTopicId === topicId && i.type === 'vision')
    .first();
}

export async function getVisionVersions(versionGroup: string): Promise<Insight[]> {
  return db.insights.filter(i => i.versionGroup === versionGroup)
    .reverse().sortBy('createdAt');
}

export async function getBaselineByTopic(topicId: string): Promise<Insight | undefined> {
  return db.insights.filter(i => i.linkedTopicId === topicId && i.type === 'baseline')
    .first();
}

export async function getAllBaselines(): Promise<Insight[]> {
  return db.insights.filter(i => i.type === 'baseline').toArray();
}

// ── 关系 CRUD ──

export async function createRelation(input: {
  sourceId: string;
  targetId: string;
  relationType: RelationType;
}): Promise<InsightRelation> {
  // 双向存储：正向 + 反向
  const forward: InsightRelation = {
    id: crypto.randomUUID(),
    sourceId: input.sourceId,
    targetId: input.targetId,
    relationType: input.relationType,
    createdAt: new Date(),
  };
  const reverseRelationType = getReverseType(input.relationType);
  const reverse: InsightRelation = {
    id: crypto.randomUUID(),
    sourceId: input.targetId,
    targetId: input.sourceId,
    relationType: reverseRelationType,
    createdAt: new Date(),
  };
  await db.insightRelations.bulkAdd([forward, reverse]);
  return forward;
}

function getReverseType(type: RelationType): RelationType {
  switch (type) {
    case 'parent': return 'child';
    case 'child': return 'parent';
    case 'sibling': return 'sibling';  // 平级对称
    case 'opposite': return 'opposite'; // 反面对称
  }
}

export async function getRelations(insightId: string): Promise<InsightRelation[]> {
  return db.insightRelations
    .filter(r => r.sourceId === insightId || r.targetId === insightId)
    .toArray();
}

export async function deleteRelation(id: string): Promise<void> {
  const rel = await db.insightRelations.get(id);
  if (!rel) return;
  // 删除正向 + 反向
  await db.insightRelations.delete(id);
  const reverse = await db.insightRelations
    .filter(r => r.sourceId === rel.targetId && r.targetId === rel.sourceId
      && r.relationType === getReverseType(rel.relationType))
    .first();
  if (reverse) await db.insightRelations.delete(reverse.id);
}

export async function checkRelationExists(sourceId: string, targetId: string): Promise<boolean> {
  const existing = await db.insightRelations
    .filter(r => r.sourceId === sourceId && r.targetId === targetId)
    .first();
  return !!existing;
}
```

## 五、SunburstWheel 实现方案

使用 **纯 SVG** 绘制，不依赖第三方图表库，根据目前约 3-6 个课题的量级，性能完全足够。

```typescript
// src/components/SunburstWheel.tsx
// Props:
//   - baselineStates: { topicId: string; topicName: string; color: string;
//       baselineStatus: 'good' | 'warn' | 'bad' | 'unset';
//       activityRatio: number; // 0-1
//       onClick?: () => void }
//   - onCenterClick?: () => void

// 算法：
// 1. 总扇区数 = 课题数
// 2. 每个扇区角度 = 360 / 课题数
// 3. 内层半径 25px → 外层半径 120px
// 4. 每层环分界：内 25-55 / 中 55-85 / 外 85-115 + 边距
// 5. 每个扇区的弧形用 SVG path arc 指令绘制
// 6. 外层用 dasharray 实现填充比例效果
```
